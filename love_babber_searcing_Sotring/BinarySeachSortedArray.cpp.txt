#include <iostream>
#include <vector>
using namespace std;

int BinarySearchLeastSortedArray(vector<int>&arr,int target){
    int s=0;
    int e=arr.size()-1;
    int mid = s+(e-s)/2;

    while(s<=e){
        if(arr[mid]== target){
            return mid;
        }
        if(arr[mid-1]==target){
            return mid-1;

        }
        if(arr[mid+1]==target){
            return mid+1;
        }
        if(target>arr[mid]){
            s=mid + 1;
        }
        else{
            e = mid - 1;
        }
    }
    return -1;
}

int main() 
{

    vector<int>arr{10,3,40,20,50,80,70};
    // cout << "Hello, World!";
    int target = 40;
    int index = BinarySearchLeastSortedArray(arr,target);
    if(index==-1){
        cout<<"Not Found"<<endl;
    }else{
        cout<<"Present "<<arr[index]<< " at - "<<index<<endl;
    }
    return 0;
}
